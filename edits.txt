/*
 * afatfs_deleteTree
 *
 * What: Public API to initiate a native recursive tree deletion.
 * Inputs: A pointer to the root object ID (resolved from previous scan).
 * Outputs: Calls the provided callback with AFATFS_RESULT_OK when the entire
 * tree is deleted and FAT clusters freed, or an error code otherwise.
 */
bool afatfs_deleteTree(const afatfsObjectId_t *root, afatfsResultCallback_t cb)
{
    afatfsFilePtr_t file = afatfs_allocateFileHandle();
    if (!file) return false;
    
    afatfs_initFileHandle(file);
    file->operation.operation = AFATFS_FILE_OPERATION_DELETE_TREE;
    file->operation.state.deleteTree.phase = AFATFS_DELETE_TREE_INITIAL;
    file->operation.state.deleteTree.callback = cb;
    file->operation.state.deleteTree.rootId = *root;
    
    return true;
}

bool afatfs_fopenChild(afatfsDirHandle_t parent, const char *displayName, afatfsCreateMode_t mode, afatfsFileCallback_t complete)
{
    // Mode behavior will be enforced in afatfs_createFileContinue
    afatfsFilePtr_t file = afatfs_allocateFileHandle();
    if (file) {
        // We inject the parent's cluster and attributes instead of relying on afatfs.currentDirectory
        file->firstCluster = parent->firstCluster;
        file->physicalSize = parent->physicalSize;
        file->logicalSize = parent->logicalSize;
        file->attrib = parent->attrib;
        char unusedNameOut[AFATFS_SHORT_FILENAME_MAX];
        
        afatfs_createFileInternal(file, displayName, FAT_FILE_ATTRIBUTE_ARCHIVE,
                                  AFATFS_FILE_MODE_READ | AFATFS_FILE_MODE_WRITE, 
                                  AFATFS_MATCH_CASE_SENSITIVE, unusedNameOut, true,
                                  complete);
    } else if (complete) {
        complete(NULL);
    }
    return file != NULL;
}

bool afatfs_mkdirChild(afatfsDirHandle_t parent, const char *displayName, afatfsCreateMode_t mode, afatfsFileCallback_t complete)
{
    afatfsFilePtr_t file = afatfs_allocateFileHandle();
    if (file) {
        file->firstCluster = parent->firstCluster;
        file->physicalSize = parent->physicalSize;
        file->logicalSize = parent->logicalSize;
        file->attrib = parent->attrib;
        char unusedNameOut[AFATFS_SHORT_FILENAME_MAX];
        
        afatfs_createFileInternal(file, displayName, FAT_FILE_ATTRIBUTE_DIRECTORY,
                                  AFATFS_FILE_MODE_READ | AFATFS_FILE_MODE_WRITE | AFATFS_FILE_MODE_CREATE, 
                                  AFATFS_MATCH_CASE_SENSITIVE, unusedNameOut, true,
                                  complete);
    } else if (complete) {
        complete(NULL);
    }
    return file != NULL;
}
